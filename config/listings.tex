  keepspaces=true,
  showstringspaces=false,

  stringstyle={},
  numbers=none,
  captionpos=b,
  xleftmargin=0.5em,
  xrightmargin=0.5em,
  aboveskip=0.8em,
  belowskip=0.8em
}
